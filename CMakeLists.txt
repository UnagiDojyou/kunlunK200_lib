cmake_minimum_required(VERSION 3.22)
project(KunlunK200 VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)

set(KUNLUN_XRE "$ENV{XRE}" CACHE PATH "External XRE 4.33 root")
if(NOT KUNLUN_XRE)
    set(KUNLUN_XRE "/usr/local/xpu-4.33.0")
endif()
set(KUNLUN_XDNN "$ENV{XDNN}" CACHE PATH "External KL1 XDNN root")
set(KUNLUN_XTDK "$ENV{XTDK_DIR}" CACHE PATH "External native x86 XTDK root")
set(KUNLUN_PREBUILT_KERNELS "" CACHE PATH "Optional directory of previously compiled kernels")
option(KUNLUN_BUILD_BENCHMARKS "Build general library and original SDK benchmarks" ON)
find_path(KUNLUN_RUNTIME_INCLUDE xpu/runtime.h PATHS "${KUNLUN_XRE}/include" NO_DEFAULT_PATH REQUIRED)
find_path(KUNLUN_XDNN_INCLUDE xpu/xdnn.h PATHS "${KUNLUN_XDNN}/include" NO_DEFAULT_PATH REQUIRED)
find_library(KUNLUN_XPURT xpurt PATHS "${KUNLUN_XRE}/lib64" "${KUNLUN_XRE}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(KUNLUN_XPUAPI xpuapi PATHS "${KUNLUN_XDNN}/so" NO_DEFAULT_PATH REQUIRED)
get_filename_component(KUNLUN_RUNTIME_LIBDIR "${KUNLUN_XPURT}" DIRECTORY)
get_filename_component(KUNLUN_XDNN_LIBDIR "${KUNLUN_XPUAPI}" DIRECTORY)
set(CMAKE_BUILD_RPATH "${KUNLUN_RUNTIME_LIBDIR};${KUNLUN_XDNN_LIBDIR}")
set(CMAKE_INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR};$ORIGIN;${KUNLUN_RUNTIME_LIBDIR};${KUNLUN_XDNN_LIBDIR}")
set(kernel_output "${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_INSTALL_LIBDIR}/kunlunk200/kernels")
file(MAKE_DIRECTORY "${kernel_output}")
foreach(name kunlun-matvec kunlun-elementwise)
    if(KUNLUN_PREBUILT_KERNELS)
        foreach(ext bin entry)
            configure_file("${KUNLUN_PREBUILT_KERNELS}/${name}.${ext}" "${kernel_output}/${name}.${ext}" COPYONLY)
        endforeach()
    else()
        if(NOT EXISTS "${KUNLUN_XTDK}/bin/clang-10")
            message(FATAL_ERROR "Set KUNLUN_XTDK (tested: 2.7.98.2 x86) or KUNLUN_PREBUILT_KERNELS")
        endif()
        add_custom_command(OUTPUT "${kernel_output}/${name}.bin" "${kernel_output}/${name}.entry"
            COMMAND ${CMAKE_COMMAND} -E env "XTDK_DIR=${KUNLUN_XTDK}"
                bash "${CMAKE_CURRENT_SOURCE_DIR}/cmake/compile-native.sh"
                "${CMAKE_CURRENT_SOURCE_DIR}/kernels/${name}.xpu" "${kernel_output}"
            DEPENDS "kernels/${name}.xpu" cmake/compile-native.sh VERBATIM)
    endif()
    list(APPEND kernel_files "${kernel_output}/${name}.bin" "${kernel_output}/${name}.entry")
endforeach()
add_custom_target(kunlun-kernels DEPENDS ${kernel_files})
add_library(kunlunk200 SHARED src/k200.cpp)
add_library(KunlunK200::kunlunk200 ALIAS kunlunk200)
add_dependencies(kunlunk200 kunlun-kernels)
target_compile_features(kunlunk200 PUBLIC cxx_std_14)
target_include_directories(kunlunk200 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE "${KUNLUN_RUNTIME_INCLUDE}" "${KUNLUN_XDNN_INCLUDE}")
target_link_libraries(kunlunk200 PRIVATE "${KUNLUN_XPUAPI}" "${KUNLUN_XPURT}" Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(kunlunk200 PRIVATE -Wall -Wextra)
set_target_properties(kunlunk200 PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_INSTALL_LIBDIR}")
if(KUNLUN_BUILD_BENCHMARKS)
    add_executable(k200-bench bench/library-bench.cpp)
    target_link_libraries(k200-bench PRIVATE KunlunK200::kunlunk200)
    foreach(pair "k200-gemm-bench|bench/k200_gemm_bench.cpp" "k200-int4-bench|bench/k200_int4_bench.cpp"
                 "k200-pipe-bench|bench/k200_pipe_bench.cpp" "k200-smoke|setup/k200-smoke.cpp" "k200-vvadd|examples/vvadd_host.cpp")
        string(REPLACE "|" ";" pair "${pair}")
        list(GET pair 0 target)
        list(GET pair 1 source)
        add_executable(${target} ${source})
        target_compile_features(${target} PRIVATE cxx_std_14)
        target_include_directories(${target} PRIVATE "${KUNLUN_RUNTIME_INCLUDE}" "${KUNLUN_XDNN_INCLUDE}")
        target_link_libraries(${target} PRIVATE "${KUNLUN_XPUAPI}" "${KUNLUN_XPURT}")
        install(TARGETS ${target} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    endforeach()
    install(TARGETS k200-bench RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS kunlunk200 EXPORT KunlunK200Targets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES ${kernel_files} DESTINATION ${CMAKE_INSTALL_LIBDIR}/kunlunk200/kernels)
install(EXPORT KunlunK200Targets NAMESPACE KunlunK200:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KunlunK200)
configure_package_config_file(cmake/KunlunK200Config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/KunlunK200Config.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KunlunK200)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/KunlunK200ConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/KunlunK200Config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/KunlunK200ConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KunlunK200)
install(PROGRAMS tools/thermal-guard.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME k200-thermal-guard)
install(FILES LICENSE THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/kunlunk200)
